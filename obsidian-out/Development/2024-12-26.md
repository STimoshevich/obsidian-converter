<!DOCTYPE html>

<head>

    <title>Draggable L Shape with Anchors</title>
    <style>
        body {
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            margin: 0;
            background-color: #f0f0f0;
        }
        svg {
            border: 1px solid #000;
            cursor: pointer;
        }
        .anchor {
            fill: red;
            stroke: black;
            stroke-width: 1px;
            cursor: move;
        }
        .hover {
            fill: rgba(0, 123, 255, 0.5);
            stroke: #28a745;
            stroke-width: 2px;
        }
    </style>
</head>
<body>
    <svg id="svgCanvas" width="800" height="600">
        <polyline id="lShape" points="100,300 100,100 300,100 300,300" fill="none" stroke="blue" stroke-width="4"/>
        <circle class="anchor" cx="100" cy="300" r="8"/>
        <circle class="anchor" cx="100" cy="100" r="8"/>
        <circle class="anchor" cx="300" cy="100" r="8"/>
        <circle class="anchor" cx="300" cy="300" r="8"/>
    </svg>
    <script>
        const svgCanvas = document.getElementById('svgCanvas');
        const anchors = document.querySelectorAll('.anchor');
        const lShape = document.getElementById('lShape');
        let selectedAnchor = null;

        svgCanvas.addEventListener('mousedown', (e) => {
            if (e.target.classList.contains('anchor')) {
                selectedAnchor = e.target;
                selectedAnchor.classList.add('hover');
            }
        });

        svgCanvas.addEventListener('mouseup', () => {
            if (selectedAnchor) {
                selectedAnchor.classList.remove('hover');
                selectedAnchor = null;
            }
        });

        svgCanvas.addEventListener('mousemove', (e) => {
            if (selectedAnchor) {
                const rect = svgCanvas.getBoundingClientRect();
                const mouseX = e.clientX - rect.left;
                const mouseY = e.clientY - rect.top;
                selectedAnchor.setAttribute('cx', mouseX);
                selectedAnchor.setAttribute('cy', mouseY);
                updateLShape();
            }
        });

        function updateLShape() {
            const points = Array.from(anchors).map(anchor => `${anchor.getAttribute('cx')},${anchor.getAttribute('cy')}`).join(' ');
            lShape.setAttribute('points', points);
        }
    </script>
</body>
</html>
