    <style>

    <canvas id="canvas" width="800" height="600"></canvas>
    <script>
        (function() {
            const canvas = document.getElementById('canvas');
            const ctx = canvas.getContext('2d');

            let anchors = [
                { x: 100, y: 300 }, // Bottom left
                { x: 100, y: 100 }, // Top left
                { x: 300, y: 100 }, // Top right
                { x: 300, y: 300 }  // Bottom right
            ];
            let draggingIndex = -1;

            function drawLines() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                ctx.beginPath();
                ctx.moveTo(anchors[0].x, anchors[0].y);
                ctx.lineTo(anchors[1].x, anchors[1].y);
                ctx.lineTo(anchors[2].x, anchors[2].y);
                ctx.lineTo(anchors[3].x, anchors[3].y);
                ctx.strokeStyle = 'blue';
                ctx.lineWidth = 4;
                ctx.stroke();
                drawAnchors();
            }

            function drawAnchors() {
                anchors.forEach(anchor => {
                    ctx.beginPath();
                    ctx.arc(anchor.x, anchor.y, 8, 0, Math.PI * 2);
                    ctx.fillStyle = 'red';
                    ctx.fill();
                    ctx.strokeStyle = 'black';
                    ctx.stroke();
                });
            }

            function getMousePos(evt) {
                const rect = canvas.getBoundingClientRect();
                return {
                    x: evt.clientX - rect.left,
                    y: evt.clientY - rect.top
                };
            }

            canvas.addEventListener('mousedown', (e) => {
                const mousePos = getMousePos(e);
                draggingIndex = anchors.findIndex(anchor => 
                    Math.hypot(anchor.x - mousePos.x, anchor.y - mousePos.y) < 8
                );
            });

            canvas.addEventListener('mouseup', () => {
                draggingIndex = -1;
            });

            canvas.addEventListener('mousemove', (e) => {
                if (draggingIndex !== -1) {
                    const mousePos = getMousePos(e);
                    anchors[draggingIndex].x = mousePos.x;
                    anchors[draggingIndex].y = mousePos.y;
                    drawLines();
                }
            });

            drawLines();
        })();
    </script>
</body>
</html>
